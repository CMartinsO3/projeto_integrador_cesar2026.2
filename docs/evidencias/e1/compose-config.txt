name: projeto_integrador_cesar20262
services:
  api:
    build:
      context: C:\dev\projeto_integrador_cesar2026.2
      dockerfile: Dockerfile
      args:
        APP_VERSION: 0.0.1-SNAPSHOT
        GIT_SHA: local-compose
    container_name: hemoflow-api
    depends_on:
      db:
        condition: service_healthy
        required: true
    environment:
      GIT_SHA: local-compose
      HEMOFLOW_ADMIN_PASSWORD: admin
      PORT: "8080"
      SPRING_DATASOURCE_PASSWORD: hemoflow_local_dev
      SPRING_DATASOURCE_URL: jdbc:postgresql://db:5432/hemoflow
      SPRING_DATASOURCE_USERNAME: hemoflow
      SPRING_PROFILES_ACTIVE: postgres
    healthcheck:
      test:
        - CMD-SHELL
        - curl -fsS http://localhost:$${PORT:-8080}/actuator/health || exit 1
      timeout: 5s
      interval: 15s
      retries: 5
      start_period: 40s
    networks:
      hemoflow-net: null
    ports:
      - mode: ingress
        target: 8080
        published: "8080"
        protocol: tcp
    restart: unless-stopped
  db:
    container_name: hemoflow-db
    environment:
      POSTGRES_DB: hemoflow
      POSTGRES_PASSWORD: hemoflow_local_dev
      POSTGRES_USER: hemoflow
    healthcheck:
      test:
        - CMD-SHELL
        - pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}
      timeout: 5s
      interval: 10s
      retries: 5
      start_period: 10s
    image: postgres:16.4-alpine
    networks:
      hemoflow-net: null
    restart: unless-stopped
    volumes:
      - type: volume
        source: hemoflow_pgdata
        target: /var/lib/postgresql/data
        volume: {}
networks:
  hemoflow-net:
    name: hemoflow-net
    driver: bridge
volumes:
  hemoflow_pgdata:
    name: hemoflow_pgdata
